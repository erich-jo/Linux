# Install Samba
sudo apt -y update && sudo apt install samba -y
samba --version

# Create Share Folder & Set Permission (Modify [File Name])
sudo mkdir [File Name]
sudo chmod 777 [File Name]

# Modify Configuration File (Modify path, [File Name])
sudo vim /etc/samba/smb.conf
---------------------------------------------------------------------------------------------------------------------------
[share]
comment = samba
path = /~/[File Name]
browsable = yes
create mask = 0770
directory mask = 0771
writable = yes
printable = no
---------------------------------------------------------------------------------------------------------------------------

# Apply Settings
sudo systemctl restart smbd
sudo systemctl status smbd

# Add Network Location on Device Accessing Share Folder (Modify [IP Address])
\\[IP Address]\share
