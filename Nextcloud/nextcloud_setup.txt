◆ MariaDB Setup

# Auto-Start MariaDB After System Reboot
sudo systemctl enable mariadb

# MariaDB Security Setting
sudo mariadb-secure-installation

# Log in to MariaDB as Root User & Enter the commands one by one
sudo mysql -u root
---------------------------------------------------------------------------------------------------------------------------
# Create New User & Grant Permission (Modify [ID])
CREATE USER '[ID]'@'localhost' IDENTIFIED BY '1.....';

# Create New Database
CREATE DATABASE IF NOT EXISTS nextcloud CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;

# Grant privileges to account for external access in MariaDB (Modify [ID])
GRANT ALL PRIVILEGES ON nextcloud.* TO '[ID]'@'localhost';
FLUSH PRIVILEGES;

#  Verify Database and User
SHOW DATABASES;
SELECT User, Host FROM mysql.user;

EXIT;
---------------------------------------------------------------------------------------------------------------------------

# Firewall Setting
sudo ufw allow 3306/tcp
sudo ufw reload

# Check MariaDB Status
sudo systemctl status mariadb

◆ Replace Default Apache Webpage with Nextcloud Webpage

# Enable Additional Required Apache Modules
sudo a2enmod dir env headers mime rewrite ssl

# Extract Nextcloud ZIP File at Downloaded Location
unzip latest.zip

# Ensure that "www-data" user is the owner of new directory before moving extracted folder to /var/www/html
sudo chown -R www-data:www-data ./nextcloud

# Move the folder to /var/www/html directory
sudo mv ./nextcloud /var/www/html

# Disable Default Apache Web Server Page
sudo a2dissite 000-default.conf
sudo a2dissite default-ssl.conf

# Ignore Apache Server Reload Warning & Create Virtual Host File for the New Site (Modify [Domain Name])
sudo vim /etc/apache2/sites-available/nextcloud.conf
---------------------------------------------------------------------------------------------------------------------------
<VirtualHost *:80>
	DocumentRoot /var/www/html/nextcloud
	ServerName [Domain Name]
	<Directory /var/www/html/nextcloud>
		Require all granted
		AllowOverride All
		Options FollowSymLinks MultiViews
		<IfModule mod_dav.c>
			Dav off
		</IfModule>
	</Directory>
</VirtualHost>
---------------------------------------------------------------------------------------------------------------------------

# Configure Scheduled Execution Command for Nextcloud Server
sudo crontab -u www-data -e

# Select option 2(vim.basic) & Add the following line
*/5 * * * * php -f /var/www/html/nextcloud/cron.php

※ Definition of the first numbers
Execution Interval - (*/5)Every 5 minutes / (*)Every hour / (*)Every day / (*)Every month / (*)Every day of the week

# Nextcloud Website → Click Administrator Profile → Administration Settings → Basic Settings → Select Cron


◆ HTTPS Setup

# Enable and generate self-signed certificates (Modify [File Name])
sudo a2enmod ssl
sudo openssl req -x509 -nodes -days 365 -newkey rsa:2048 -keyout /etc/ssl/private/[File Name].key -out /etc/ssl/certs/[File Name].crt

# Add the lines at the virtual host file (Modify [Domain Name], [File Name])
sudo vim /etc/apache2/sites-available/nextcloud.conf
---------------------------------------------------------------------------------------------------------------------------
<VirtualHost *:443>
	DocumentRoot /var/www/html/nextcloud/
	ServerName [Domain Name]

	SSLEngine on
	SSLCertificateFile /etc/ssl/certs/[File Name].crt
	SSLCertificateKeyFile /etc/ssl/private/[File Name].key

	<Directory /var/www/html/nextcloud/>
		Require all granted
		AllowOverride All
		Options FollowSymLinks MultiViews

		<IfModule mod_dav.c>
			Dav off
		</IfModule>
	</Directory>
</VirtualHost>
---------------------------------------------------------------------------------------------------------------------------

# Modify Nextcloud config.php (Modify [Tailscale IP Address], [Domain Name])
sudo vim /var/www/html/nextcloud/config/config.php
---------------------------------------------------------------------------------------------------------------------------
trusted_domains' => 
  array (
    0 => 'localhost',
    1 => '[Tailscale IP Address]',
    2 => '[Domain Name]',
  ),
...
'overwrite.cli.url' => 'https://[Domain Name]',
---------------------------------------------------------------------------------------------------------------------------
