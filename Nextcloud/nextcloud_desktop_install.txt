◆ Install Nextcloud Desktop
sudo add-apt-repository ppa:nextcloud-devs/client
sudo apt -y update && sudo apt install nextcloud-desktop -y


◆ Delete Nextcloud Desktop
sudo apt purge --auto-remove nextcloud-desktop
sudo add-apt-repository --remove ppa:nextcloud-devs/client
sudo apt -y update && sudo rm -rf ~/.config/Nextcloud ~/.local/share/data/Nextcloud
