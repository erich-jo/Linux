# Install SSH server for remote access
sudo apt update && sudo apt upgrade -y
sudo apt install openssh-server -y
sudo systemctl enable --now ssh

# Install Firewall
sudo apt -y install ufw

# Enable Firewall & Allow Web Traffic
sudo ufw allow OpenSSH
sudo ufw allow ssh
sudo ufw allow http
sudo ufw allow https
sudo ufw allow 80/tcp
sudo ufw allow 443/tcp
sudo ufw enable
sudo ufw status

# Access from another computer (Modify [ID], [IP Address])
ssh [ID]@[IP Address]

# Check the list of network processes using port 80
sudo lsof -i :80
